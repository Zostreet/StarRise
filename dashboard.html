/* ================================================== MY OPPORTUNITIES ================================================== Shows opportunities where the current member: 1. Created the opportunity OR 2. Applied to the opportunity Application status is preserved: - pending - accepted - declined - etc. This means an opportunity appears immediately after the member applies. It does NOT require acceptance first. ================================================== */ async function loadMyOpportunities(user){ setStatus( myOpportunitiesStatus, 'Loading your opportunities...' ); myOpportunitiesContainer.innerHTML = ''; try{ /* ---------------------------------------------- CREATED OPPORTUNITIES ---------------------------------------------- */ const createdResult = await supabaseClient .from('opportunities') .select('*') .eq( 'creator_id', user.id ) .order( 'created_at', { ascending:false } ); if(createdResult.error){ throw createdResult.error; } const created = ( createdResult.data || [] ).map( opportunity => ({ ...opportunity, member_status:'created', member_role:'creator' }) ); /* ---------------------------------------------- APPLIED OPPORTUNITIES ---------------------------------------------- IMPORTANT: Do NOT filter by status here. Pending applications must also appear in My Opportunities. Previously this was: .eq('status','accepted') That was the bug. ---------------------------------------------- */ const applicationsResult = await supabaseClient .from('opportunity_applications') .select(` opportunity_id, applicant_id, status, created_at, opportunities (*) `) .eq( 'applicant_id', user.id ); if(applicationsResult.error){ throw applicationsResult.error; } const applied = ( applicationsResult.data || [] ) .filter( item => item.opportunities ) .map( item => ({ ...item.opportunities, member_status: item.status || 'pending', member_role:'applicant', application_created_at: item.created_at }) ); /* ---------------------------------------------- COMBINE CREATED + APPLIED ---------------------------------------------- */ const combined = [ ...created, ...applied ]; /* ---------------------------------------------- REMOVE DUPLICATES ---------------------------------------------- If the member created an opportunity and also somehow has an application record for it, keep the creator version. ---------------------------------------------- */ const uniqueMap = new Map(); combined.forEach( opportunity => { const existing = uniqueMap.get( opportunity.id ); if(!existing){ uniqueMap.set( opportunity.id, opportunity ); return; } /* Creator status takes priority over applicant status. */ if( existing.member_role !== 'creator' && opportunity.member_role === 'creator' ){ uniqueMap.set( opportunity.id, opportunity ); } } ); const unique = Array.from( uniqueMap.values() ); /* ---------------------------------------------- NEWEST FIRST ---------------------------------------------- */ unique.sort( (a,b) => { const dateA = new Date( a.created_at || a.application_created_at || 0 ).getTime(); const dateB = new Date( b.created_at || b.application_created_at || 0 ).getTime(); return dateB - dateA; } ); /* ---------------------------------------------- NOTHING FOUND ---------------------------------------------- */ if(!unique.length){ setStatus( myOpportunitiesStatus, 'You have not created or applied to any opportunities yet.' ); myOpportunitiesContainer.innerHTML = '
' + 'Create an opportunity or apply to one to see it here.' + '
'; return; } /* ---------------------------------------------- SUCCESS ---------------------------------------------- */ setStatus( myOpportunitiesStatus, unique.length === 1 ? '1 opportunity connected to you.' : unique.length + ' opportunities connected to you.', 'success' ); myOpportunitiesContainer.innerHTML = unique .map( opportunity => renderMyOpportunity( opportunity ) ) .join(''); }catch(error){ console.error( 'My Opportunities error:', error ); setStatus( myOpportunitiesStatus, 'Unable to load your opportunities: ' + ( error.message || 'Unknown error' ), 'error' ); } } /* ================================================== RENDER MY OPPORTUNITY ================================================== */ function renderMyOpportunity( opportunity ){ const type = String( opportunity.type || 'opportunity' ).toLowerCase(); /* ---------------------------------------------- MEMBER STATUS ---------------------------------------------- */ const memberStatus = opportunity.member_status || opportunity.status || 'pending'; /* ---------------------------------------------- MEMBER ROLE ---------------------------------------------- */ const memberRole = opportunity.member_role || 'applicant'; /* ---------------------------------------------- LOCATION ---------------------------------------------- */ const location = opportunity.location && String( opportunity.location ).trim() ? opportunity.location : opportunity.is_remote ? 'Remote' : 'Location to be announced'; /* ---------------------------------------------- DATE ---------------------------------------------- */ const eventDate = opportunity.event_date ? formatDate( opportunity.event_date ) : 'Date to be announced'; /* ---------------------------------------------- COMPENSATION ---------------------------------------------- */ const compensation = opportunity.compensation !== undefined && opportunity.compensation !== null && String( opportunity.compensation ).trim() ? opportunity.compensation : 'Not specified'; /* ---------------------------------------------- DESCRIPTION ---------------------------------------------- */ const description = opportunity.description || ''; /* ---------------------------------------------- STATUS DISPLAY ---------------------------------------------- */ let statusLabel = memberStatus; if( memberRole === 'creator' ){ statusLabel = 'created'; } /* ---------------------------------------------- ROLE DISPLAY ---------------------------------------------- */ const roleLabel = memberRole === 'creator' ? 'Creator' : 'Applicant'; return `
${escapeHtml(type)}
${escapeHtml( opportunity.title || 'Opportunity' )}
${escapeHtml( description )}
📍 ${escapeHtml( location )}
📅 ${escapeHtml( eventDate )}
💰 ${escapeHtml( compensation )}
👤 ${escapeHtml( roleLabel )}
${escapeHtml( statusLabel )}

View Opportunity
`; }
