async function loadMyOpportunities(user) { setStatus(myOpportunitiesStatus, 'Loading your opportunities...'); myOpportunitiesContainer.innerHTML = ''; try { const createdResult = await supabaseClient .from('opportunities') .select('*') .eq('creator_id', user.id) .order('created_at', { ascending: false }); if (createdResult.error) throw createdResult.error; const created = (createdResult.data || []).map(opportunity => ({ ...opportunity, member_status: 'created', member_role: 'creator' })); const applicationsResult = await supabaseClient .from('opportunity_applications') .select(` opportunity_id, applicant_id, status, created_at, opportunities (*) `) .eq('applicant_id', user.id); if (applicationsResult.error) throw applicationsResult.error; const applied = (applicationsResult.data || []) .filter(item => item.opportunities) .map(item => ({ ...item.opportunities, member_status: item.status || 'pending', member_role: 'applicant', application_created_at: item.created_at })); const combined = [...created, ...applied]; const uniqueMap = new Map(); combined.forEach(opportunity => { const existing = uniqueMap.get(opportunity.id); if (!existing) { uniqueMap.set(opportunity.id, opportunity); return; } if ( existing.member_role !== 'creator' && opportunity.member_role === 'creator' ) { uniqueMap.set(opportunity.id, opportunity); } }); const unique = Array.from(uniqueMap.values()); unique.sort((a, b) => { const dateA = new Date( a.application_created_at || a.created_at || 0 ).getTime(); const dateB = new Date( b.application_created_at || b.created_at || 0 ).getTime(); return dateB - dateA; }); if (!unique.length) { setStatus( myOpportunitiesStatus, 'You have not created or applied to any opportunities yet.' ); myOpportunitiesContainer.innerHTML = `
Create an opportunity or apply to one to see it here.
`; return; } setStatus( myOpportunitiesStatus, unique.length === 1 ? '1 opportunity connected to you.' : `${unique.length} opportunities connected to you.`, 'success' ); myOpportunitiesContainer.innerHTML = unique .map(opportunity => renderMyOpportunity(opportunity)) .join(''); } catch (error) { console.error('My Opportunities error:', error); setStatus( myOpportunitiesStatus, 'Unable to load your opportunities: ' + (error.message || 'Unknown error'), 'error' ); } } function renderMyOpportunity(opportunity) { const type = String( opportunity.type || 'opportunity' ).toLowerCase(); const memberStatus = opportunity.member_status || opportunity.status || 'pending'; const memberRole = opportunity.member_role || 'applicant'; const location = opportunity.location && String(opportunity.location).trim() ? opportunity.location : opportunity.is_remote ? 'Remote' : 'Location to be announced'; const eventDate = opportunity.event_date ? formatDate(opportunity.event_date) : 'Date to be announced'; const compensation = opportunity.compensation !== undefined && opportunity.compensation !== null && String(opportunity.compensation).trim() ? opportunity.compensation : 'Not specified'; const description = opportunity.description || ''; const statusLabel = memberRole === 'creator' ? 'created' : memberStatus; const roleLabel = memberRole === 'creator' ? 'Creator' : 'Applicant'; return `
${escapeHtml(type)}

${escapeHtml( opportunity.title || 'Opportunity' )}

${escapeHtml(description)}

📍 ${escapeHtml(location)}
📅 ${escapeHtml(eventDate)}
💰 ${escapeHtml(compensation)}
👤 ${escapeHtml(roleLabel)}
${escapeHtml(statusLabel)}
`; }
